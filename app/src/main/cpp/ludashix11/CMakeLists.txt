cmake_minimum_required(VERSION 3.22.1)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(LudashiX11 LANGUAGES C CXX)
endif()
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS core/*.cpp core/io/*.cpp core/cursor/*.cpp base/*.cpp)
set(NATIVE_SOURCES src/native_drawable.cpp src/native_screen.cpp src/native_extensions.cpp src/native_glx.cpp src/native_randr.cpp src/native_compat.cpp src/native_xinput.cpp src/native_displayx_bridge.cpp)
if(ANDROID)
    if(NOT TARGET vulkan_renderer)
        message(FATAL_ERROR "LudashiX11 requires the app Vulkan renderer target")
    endif()
    get_target_property(RENDERER_SOURCES vulkan_renderer SOURCES)
    list(FILTER RENDERER_SOURCES EXCLUDE REGEX "vulkan_jni\\.cpp$|lsfg_jni\\.c$")
    set(ABS_RENDERER_SOURCES)
    foreach(source IN LISTS RENDERER_SOURCES)
        if(IS_ABSOLUTE "${source}")
            list(APPEND ABS_RENDERER_SOURCES "${source}")
        else()
            list(APPEND ABS_RENDERER_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../${source}")
        endif()
    endforeach()
    add_library(ludashix11 SHARED ${CORE_SOURCES} ${NATIVE_SOURCES} ${ABS_RENDERER_SOURCES}
        src/native_renderer.cpp src/native_egl_renderer.cpp src/native_displayx_renderer.cpp src/native_jni.cpp)
    target_compile_definitions(ludashix11 PRIVATE LUDASHI_NATIVE_X11=1)
    add_dependencies(ludashix11 vulkan_renderer)
    target_include_directories(ludashix11 PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}/../winlator/renderer/vulkan
        ${CMAKE_CURRENT_BINARY_DIR}/..)
    target_link_libraries(ludashix11 PRIVATE android log vulkan EGL GLESv2 adrenotools dl atomic wfg)
else()
    add_executable(ludashix11_host ${CORE_SOURCES} ${NATIVE_SOURCES} tests/host_backend.cpp)
    set(LUDASHI_TARGET ludashix11_host)
    target_include_directories(ludashix11_host PRIVATE tests/stubs)
    target_link_libraries(ludashix11_host PRIVATE pthread)
    target_compile_definitions(ludashix11_host PRIVATE LUDASHI_HOST_TEST=1)
    add_executable(ludashix11_renderer_test tests/test_renderer.cpp src/native_renderer.cpp
        src/native_drawable.cpp core/drawable.cpp)
    target_include_directories(ludashix11_renderer_test PRIVATE tests/renderer_stubs tests/stubs . include third_party)
    target_compile_features(ludashix11_renderer_test PRIVATE cxx_std_20)
    target_compile_options(ludashix11_renderer_test PRIVATE -include ${CMAKE_CURRENT_SOURCE_DIR}/include/common.h)
    target_link_libraries(ludashix11_renderer_test PRIVATE pthread)
endif()
if(ANDROID)
    set(LUDASHI_TARGET ludashix11)
endif()
set_target_properties(${LUDASHI_TARGET} PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
target_include_directories(${LUDASHI_TARGET} PRIVATE . include third_party)
target_compile_definitions(${LUDASHI_TARGET} PRIVATE FMT_HEADER_ONLY=1)
target_compile_options(${LUDASHI_TARGET} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-include;${CMAKE_CURRENT_SOURCE_DIR}/include/common.h>"
    -Wall -Wextra -Wno-unused-parameter -Wno-missing-field-initializers -fvisibility=hidden)
